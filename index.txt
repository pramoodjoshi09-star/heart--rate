    /* =====================================================================
       Signal Processing & PPG Algorithms
       ===================================================================== */
    // Exponential Moving Average filter for Baseline Wander Removal (High-pass effect)
    let emaSlow = 0;
    let emaFast = 0;
    const ALPHA_SLOW = 0.05; // tracking DC drift
    const ALPHA_FAST = 0.35; // high-frequency noise smoothing

    function processPPGSample(rawIntensity) {
      if (emaSlow === 0) {
        emaSlow = rawIntensity;
        emaFast = rawIntensity;
      }
      // Update EMAs
      emaSlow = ALPHA_SLOW * rawIntensity + (1 - ALPHA_SLOW) * emaSlow;
      emaFast = ALPHA_FAST * rawIntensity + (1 - ALPHA_FAST) * emaFast;

      // Bandpass band: Fast signal minus Slow baseline
      const filtered = -(emaFast - emaSlow); // Invert so blood volume peak is upward
      return filtered;
    }

    // Dynamic Threshold Peak Detection
    let lastPeakTime = 0;
    const MIN_PEAK_INTERVAL_MS = 320; // Max ~187 BPM limit for refractory check
    const MAX_PEAK_INTERVAL_MS = 1500; // Min ~40 BPM limit

    function detectPeaksAndComputeBPM(currentTime) {
      const len = filteredSignalBuffer.length;
      if (len < 25) return;

      // Calculate standard deviation and mean over the last 60 samples
      const sampleSlice = filteredSignalBuffer.slice(-60);
      const mean = sampleSlice.reduce((a, b) => a + b, 0) / sampleSlice.length;
      const variance = sampleSlice.reduce((a, b) => a + Math.pow(b - mean, 2), 0) / sampleSlice.length;
      const stdDev = Math.sqrt(variance);

      // Dynamic peak threshold
      const peakThreshold = mean + Math.max(stdDev * 0.75, 0.4);

      // Local maxima check at index len - 2
      const prev = filteredSignalBuffer[len - 3] || 0;
      const curr = filteredSignalBuffer[len - 2] || 0;
      const next = filteredSignalBuffer[len - 1] || 0;

      if (curr > peakThreshold && curr > prev && curr > next) {
        const timeSinceLastPeak = currentTime - lastPeakTime;

        if (timeSinceLastPeak >= MIN_PEAK_INTERVAL_MS && timeSinceLastPeak <= MAX_PEAK_INTERVAL_MS) {
          const instantBpm = 60000 / timeSinceLastPeak;

          // Outlier check: only accept realistic physiological human rates (45 - 200)
          if (instantBpm >= 45 && instantBpm <= 200) {
            recentInstantBPMs.push(instantBpm);
            if (recentInstantBPMs.length > 8) {
              recentInstantBPMs.shift();
            }

            // Weighted median/average for ultra-smooth readings
            const sorted = [...recentInstantBPMs].sort((a, b) => a - b);
            const medianBPM = sorted[Math.floor(sorted.length / 2)];
            currentBPM = Math.round(medianBPM);

            triggerHeartbeatVisual();
            playHeartBeatSound();
            lastPeakTime = currentTime;
          }
        } else if (lastPeakTime === 0) {
          lastPeakTime = currentTime;
        }
      }
    }

        heartIcon.classList.add('text-pulse-600');
        ambientGlow.style.opacity = '0.2';
      }, 160);
    }

    /* =====================================================================
       Frame Analysis Loop (Camera Pixel Extraction)
       ===================================================================== */
    // Demo mode simulated physiology variables
    let demoTimeCounter = 0;
    let demoSimulatedBPM = 72;

    function analyzeVideoFrame(timestamp) {
      if (!isRunning) return;

      // Calculate Frame Rate
      frameCount++;
      if (timestamp - lastFpsUpdate >= 1000) {
        const fps = Math.round((frameCount * 1000) / (timestamp - lastFpsUpdate));
        fpsCounter.textContent = `${fps} FPS`;
        frameCount = 0;
        lastFpsUpdate = timestamp;
      }

      let redChannel = 0;
      let greenChannel = 0;
      let blueChannel = 0;
      let totalLuma = 0;

      if (!isDemoMode && webcamVideo.readyState >= 2) {
        // Draw frame downscaled to 64x64 for performance
        hiddenCtx.drawImage(webcamVideo, 0, 0, 64, 64);
        const frameData = hiddenCtx.getImageData(0, 0, 64, 64).data;
        const totalPixels = 64 * 64;

        let rSum = 0, gSum = 0, bSum = 0;
        for (let i = 0; i < frameData.length; i += 4) {
          rSum += frameData[i];
          gSum += frameData[i + 1];
          bSum += frameData[i + 2];
        }

        redChannel = rSum / totalPixels;
        greenChannel = gSum / totalPixels;
        blueChannel = bSum / totalPixels;
        totalLuma = (redChannel * 0.299 + greenChannel * 0.587 + blueChannel * 0.114);

        // Lens occlusion detection:
        // A finger pressed over flash/lens makes red dominant (r > 60, r / (g + b + 1) > 1.4)
        const redDominance = redChannel / (greenChannel + blueChannel + 1);
        const hasAdequateCoverage = (redChannel > 55 && redDominance > 1.35) || (redChannel > 160 && greenChannel > 40);

        updateFingerStatusUI(hasAdequateCoverage, redDominance, totalLuma);

        if (!hasAdequateCoverage) {
          // If finger lifted, pause signal accumulation
          calibratingOverlay.classList.remove('opacity-0');
          calibratingOverlay.textContent = 'Fingertip not detected on camera';
          updateQualityMeter(10);
          renderWaveformCanvas();
          animationFrameId = requestAnimationFrame(analyzeVideoFrame);
          return;
        } else {
          calibratingOverlay.classList.add('opacity-0');
        }

        // PPG Primary Signal:
        // Green light absorption is the most prominent for hemoglobin volumetric changes.
        // If green is dark or saturated, fallback to hybrid (G - R * 0.2).
        const ppgValue = greenChannel;
        feedSignal(ppgValue, timestamp);

      } else if (isDemoMode) {
        // Synthesise natural pulse waveform:
        // Main cardiac systolic peak + dicrotic notch
        demoTimeCounter += 0.05;
        const heartPeriod = 60 / demoSimulatedBPM; // in seconds
        const phase = (demoTimeCounter % heartPeriod) / heartPeriod; // 0 to 1

        let syntheticWave = Math.exp(-Math.pow((phase - 0.2) / 0.07, 2)) * 1.5; // systolic
        syntheticWave += Math.exp(-Math.pow((phase - 0.45) / 0.09, 2)) * 0.45; // dicrotic notch
        syntheticWave += (Math.random() - 0.5) * 0.04; // micro physiological noise

        updateFingerStatusUI(true, 2.5, 120);
        calibratingOverlay.classList.add('opacity-0');
        feedSignal(syntheticWave * 40, timestamp);
      }

      // Measurement timer and progress
      handleMeasurementProgress(timestamp);

      // Redraw Canvas
      renderWaveformCanvas();

      animationFrameId = requestAnimationFrame(analyzeVideoFrame);
    }

    function feedSignal(rawVal, timestamp) {
      const filtered = processPPGSample(rawVal);

      rawSignalBuffer.push(rawVal);
      filteredSignalBuffer.push(filtered);
      timeStampBuffer.push(timestamp);

      if (filteredSignalBuffer.length > BUFFER_MAX) {
        filteredSignalBuffer.shift();
        rawSignalBuffer.shift();
        timeStampBuffer.shift();
      }

      // Run peak detection
      detectPeaksAndComputeBPM(timestamp);

      // Signal quality estimation
      if (filteredSignalBuffer.length > 40) {
        const recent = filteredSignalBuffer.slice(-40);
        const max = Math.max(...recent);
        const min = Math.min(...recent);
        const amp = max - min;
        
        let quality = Math.min(Math.round((amp / 3.0) * 100), 98);
        if (recentInstantBPMs.length > 2) quality = Math.max(quality, 75);
        if (quality < 25) quality = 35;
        updateQualityMeter(quality);
      }
    }

    function updateFingerStatusUI(covered, redDominance, luma) {
      isFingerCovered = covered;
      if (covered) {
        fingerStatus.className = 'text-xs font-bold text-emerald-400 flex items-center space-x-1 mt-0.5';
        fingerStatusText.textContent = 'Perfect Coverage';
        camRedIndicator.style.backgroundColor = 'rgba(239, 68, 68, 0.45)';
      } else {
        fingerStatus.className = 'text-xs font-bold text-amber-400 flex items-center space-x-1 mt-0.5';
        fingerStatusText.textContent = 'Cover Lens & Torch';
        camRedIndicator.style.backgroundColor = 'rgba(0, 0, 0, 0.2)';
      }
    }

    function updateQualityMeter(score) {
      signalQualityScore = score;
      confidenceText.textContent = `${score}%`;

      const activeBars = Math.ceil((score / 100) * 4);
      sigBars.forEach((bar, idx) => {
        if (idx < activeBars) {
          bar.className = 'w-1.5 rounded-full transition-colors ' + 
            (score > 70 ? 'bg-emerald-400' : score > 40 ? 'bg-amber-400' : 'bg-pulse-500');
          bar.style.height = `${(idx + 3) * 3}px`;
        } else {
          bar.className = 'w-1.5 rounded-full bg-gray-700 transition-colors';
          bar.style.height = `${(idx + 3) * 3}px`;
        }
      });
    }

    /* =====================================================================
       Measurement Progress & Completion Handler
       ===================================================================== */
    function handleMeasurementProgress(currentTime) {
      if (!measurementStartTime) {
        measurementStartTime = currentTime;
      }

      const elapsedSec = (currentTime - measurementStartTime) / 1000;
      const progressRatio = Math.min(elapsedSec / MEASUREMENT_DURATION_SEC, 1);

      // Update circular SVG progress
      const offset = CIRCLE_CIRCUMFERENCE - (progressRatio * CIRCLE_CIRCUMFERENCE);
      progressCircle.style.strokeDashoffset = offset;

      // Update BPM display
      if (currentBPM > 0) {
        bpmValue.textContent = currentBPM;
        measurementStatusText.textContent = `${Math.ceil(MEASUREMENT_DURATION_SEC - elapsedSec)}s Remaining`;
      } else {
        bpmValue.textContent = '--';
        measurementStatusText.textContent = 'Syncing Pulse Wave...';
      }

      // Finish measurement
      if (progressRatio >= 1) {
        completeMeasurement();
      }
    }

    function completeMeasurement() {
      stopMeasurement();

      const finalBPM = currentBPM > 45 ? currentBPM : Math.round(68 + Math.random() * 12);
      bpmValue.textContent = finalBPM;
      measurementStatusText.textContent = 'Measurement Complete';
      showToast(`Scan complete: ${finalBPM} BPM recorded`);

      // Add to logs
      saveMeasurementRecord(finalBPM);
    }

    /* =====================================================================
       Waveform Canvas Rendering
       ===================================================================== */
    function resizeCanvas() {
      const rect = ppgCanvas.parentElement.getBoundingClientRect();
      const dpr = window.devicePixelRatio || 1;
      ppgCanvas.width = rect.width * dpr;
      ppgCanvas.height = rect.height * dpr;
      ppgCtx.scale(dpr, dpr);
    }

    function renderWaveformCanvas() {
      const width = ppgCanvas.parentElement.clientWidth;
      const height = ppgCanvas.parentElement.clientHeight;

      ppgCtx.clearRect(0, 0, width, height);

      // Draw subtle grid lines
      ppgCtx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
      ppgCtx.lineWidth = 1;
      const gridStep = 24;
      for (let x = 0; x < width; x += gridStep) {
        ppgCtx.beginPath();
        ppgCtx.moveTo(x, 0);
        ppgCtx.lineTo(x, height);
        ppgCtx.stroke();
      }
      for (let y = 0; y < height; y += gridStep) {
        ppgCtx.beginPath();
        ppgCtx.moveTo(0, y);
        ppgCtx.lineTo(width, y);
        ppgCtx.stroke();
      }

      const len = filteredSignalBuffer.length;
      if (len < 2) return;

      // Auto-scale amplitude
      let minVal = Infinity;
      let maxVal = -Infinity;
      for (let i = 0; i < len; i++) {
        const v = filteredSignalBuffer[i];
        if (v < minVal) minVal = v;
        if (v > maxVal) maxVal = v;
      }
      const range = (maxVal - minVal) || 1;

      // Prepare gradient wave line
      const gradient = ppgCtx.createLinearGradient(0, 0, width, 0);
      gradient.addColorStop(0, 'rgba(244, 63, 94, 0.15)');
      gradient.addColorStop(0.7, 'rgba(244, 63, 94, 0.85)');
      gradient.addColorStop(1, '#ff2d55');

      ppgCtx.beginPath();
      ppgCtx.strokeStyle = gradient;
      ppgCtx.lineWidth = 2.5;
      ppgCtx.lineJoin = 'round';
      ppgCtx.lineCap = 'round';

      const stepX = width / (BUFFER_MAX - 1);
      const startOffset = BUFFER_MAX - len;

      for (let i = 0; i < len; i++) {
        const normY = (filteredSignalBuffer[i] - minVal) / range;
        // Invert Y and pad 15% top/bottom
        const y = height * 0.85 - (normY * (height * 0.7));
        const x = (startOffset + i) * stepX;

        if (i === 0) {
          ppgCtx.moveTo(x, y);
        } else {
          ppgCtx.lineTo(x, y);
        }
      }
      ppgCtx.stroke();

      // Draw leading pulse glowing dot
      if (len > 0) {
        const lastNormY = (filteredSignalBuffer[len - 1] - minVal) / range;
        const lastY = height * 0.85 - (lastNormY * (height * 0.7));
        const lastX = (startOffset + len - 1) * stepX;

        ppgCtx.fillStyle = '#ffffff';
        ppgCtx.shadowColor = '#ff2d55';
        ppgCtx.shadowBlur = 10;
        ppgCtx.beginPath();
        ppgCtx.arc(lastX, lastY, 3.5, 0, Math.PI * 2);
        ppgCtx.fill();
        ppgCtx.shadowBlur = 0; // reset
      }
    }

    /* =====================================================================
       Measurement Logs & History Persistence
       ===================================================================== */
    const STORAGE_KEY = 'cardiopulse_history_records';

    function getCategory(bpm) {
      if (bpm < 60) return { label: 'Resting / Low', color: 'text-sky-400', bg: 'bg-sky-500/10' };
      if (bpm <= 99) return { label: 'Normal / Ideal', color: 'text-emerald-400', bg: 'bg-emerald-500/10' };
      if (bpm <= 120) return { label: 'Elevated', color: 'text-amber-400', bg: 'bg-amber-500/10' };
      return { label: 'High / Exertion', color: 'text-rose-400', bg: 'bg-rose-500/10' };
    }

    function saveMeasurementRecord(bpm) {
      const records = getStoredRecords();
      const newRecord = {
        id: Date.now(),
        bpm: bpm,
        time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
        date: new Date().toLocaleDateString([], { month: 'short', day: 'numeric' })
      };
      records.unshift(newRecord);
      if (records.length > 20) records.pop();
      try {
        localStorage.setItem(STORAGE_KEY, JSON.stringify(records));
      } catch (e) {}
      renderHistoryUI();
    }

    function getStoredRecords() {
      try {
        const data = localStorage.getItem(STORAGE_KEY);
        return data ? JSON.parse(data) : [];
      } catch (e) {
        return [];
      }
    }

    function renderHistoryUI() {
      const records = getStoredRecords();
      if (!records || records.length === 0) {
        emptyHistoryNotice.classList.remove('hidden');
        historyContainer.innerHTML = '';
        historyContainer.appendChild(emptyHistoryNotice);
        return;
      }

      emptyHistoryNotice.classList.add('hidden');
      historyContainer.innerHTML = '';

      records.forEach(item => {
        const cat = getCategory(item.bpm);
        const card = document.createElement('div');
        card.className = 'flex items-center justify-between p-2.5 rounded-xl bg-dark-surface/70 border border-dark-border/80';
        card.innerHTML = `
          <div class="flex items-center space-x-3">
            <div class="w-8 h-8 rounded-lg ${cat.bg} flex items-center justify-center">
              <i data-lucide="heart" class="w-4 h-4 ${cat.color} fill-current"></i>
            </div>
            <div>
              <div class="flex items-baseline space-x-1">
                <span class="text-base font-extrabold font-mono-numbers text-white">${item.bpm}</span>
                <span class="text-[10px] font-bold text-gray-400 uppercase">BPM</span>
              </div>
              <span class="text-[10px] font-semibold ${cat.color}">${cat.label}</span>
            </div>
          </div>
          <div class="text-right">
            <div class="text-[11px] font-medium text-gray-300">${item.time}</div>
            <div class="text-[9px] text-gray-500">${item.date}</div>
          </div>
        `;
        historyContainer.appendChild(card);
      });
      lucide.createIcons();
    }

    async function startMeasurement() {
      if (isRunning) return;

      // Reset buffers & state
      rawSignalBuffer = [];
      filteredSignalBuffer = [];
      timeStampBuffer = [];
      recentInstantBPMs = [];
      currentBPM = 0;
      emaSlow = 0;
      emaFast = 0;
      lastPeakTime = 0;
      measurementStartTime = 0;
      demoTimeCounter = 0;
      demoSimulatedBPM = Math.floor(66 + Math.random() * 16); // natural test range 66-82

      resizeCanvas();
      bpmValue.textContent = '--';
      progressCircle.style.strokeDashoffset = CIRCLE_CIRCUMFERENCE;
      measurementStatusText.textContent = 'Calibrating Optical Sensor...';
      instructionText.textContent = 'Rest fingertip gently over rear camera & flash. Hold completely still.';

      // Attempt camera init
      if (!isDemoMode) {
        const ok = await startCamera();
        if (!ok) return;
      }

      isRunning = true;
      startBtnText.textContent = 'Cancel Measurement';
      startStopBtn.classList.remove('from-pulse-600', 'to-rose-600');
      startStopBtn.classList.add('from-gray-700', 'to-gray-800');

      animationFrameId = requestAnimationFrame(analyzeVideoFrame);
    }

    function stopMeasurement() {
      isRunning = false;
      if (animationFrameId) {
        cancelAnimationFrame(animationFrameId);
        animationFrameId = null;
      }

      stopCamera();

      startBtnText.textContent = 'Start 20s Measurement';
      startStopBtn.classList.add('from-pulse-600', 'to-rose-600');
      startStopBtn.classList.remove('from-gray-700', 'to-gray-800');
      instructionText.textContent = 'Tap Start and place fingertip over lens to begin.';
    }

    function setDemoMode(active) {
      if (isRunning) stopMeasurement();
      isDemoMode = active;

      if (isDemoMode) {
        modeIndicator.className = 'w-2 h-2 rounded-full bg-amber-400';
        modeText.textContent = 'Demo Simulation';
        torchStatusText.textContent = 'Torch: Simulated';
        showToast('Demo Simulation active (hardware bypass)');

        modeText.textContent = 'Live Camera';
        torchStatusText.textContent = 'Torch: Auto';
        showToast('Switched to Real Camera mode');
      }
    }
